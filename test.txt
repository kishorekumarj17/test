import re
from typing import Literal

from pydantic import BaseModel, field_validator, model_validator

HUNK_ID_RE = re.compile(r"^\d+:\d+$")


class ReviewDecision(BaseModel):
    action: Literal["all", "none", "select"]
    accepted: list[str] = []

    @field_validator("accepted")
    @classmethod
    def check_ids(cls, ids: list[str]) -> list[str]:
        bad = [i for i in ids if not HUNK_ID_RE.match(i)]
        if bad:
            raise ValueError(f"Invalid hunk ids: {bad}. Expected '<file>:<hunk>', e.g. '0:2'")
        return sorted(set(ids), key=lambda i: tuple(map(int, i.split(":"))))

    @model_validator(mode="after")
    def check_action(self):
        if self.action == "select" and not self.accepted:
            raise ValueError("Select at least one hunk, or use action 'none'")
        if self.action != "select":
            self.accepted = []
        return self

import os

WAITING_FOR_REVIEW = "waiting_for_review"
RUN_ID_RE = re.compile(r"^[\w-]+$")
# e.g. C:\...\Migration-Skill-Factory\backend\output\coder_runs   (put it in app/.env)
CODER_RUNS_ROOT = os.getenv("CODER_RUNS_ROOT")

# Extend the ARTIFACTS dict from the assessment step:
ARTIFACTS: dict[str, tuple[str, str]] = {
    "skill": ("repository_skill.md", "text/markdown; charset=utf-8"),
    "file-analysis": ("file_analysis.json", "application/json"),
    "reading-plan": ("reading_plan.json", "application/json"),
    # coder run files
    "diff": ("full.diff", "text/plain; charset=utf-8"),
    "summary": ("summary.txt", "text/plain; charset=utf-8"),
    "accepted-patch": ("accepted.patch", "text/plain; charset=utf-8"),
}
CODER_ARTIFACTS = {"diff", "summary", "accepted-patch"}


def artifact_dir(job, name: str) -> Path:
    """Skill files live in job.output_path; coder files in coder_runs/<run_id>."""
    if name in CODER_ARTIFACTS:
        if not job.coder_run_id:
            raise HTTPException(409, "This job has no coder run yet")
        if not RUN_ID_RE.match(job.coder_run_id):
            raise HTTPException(500, "Invalid coder_run_id on job")
        if CODER_RUNS_ROOT:
            return Path(CODER_RUNS_ROOT) / job.coder_run_id
    if not job.output_path:
        raise HTTPException(409, "This job has no output yet")
    return Path(job.output_path)

    if job is None:
        raise HTTPException(404, "Job not found")
    if (job.job_type or "").lower() != "code":
        raise HTTPException(409, "Only code jobs can be reviewed")
    if job.status != WAITING_FOR_REVIEW:
        raise HTTPException(409, f"Job is '{job.status}', not waiting for review")

    job.review_decision = decision.model_dump()
    job.status = "queued"                     # worker picks it up again -> phase 2
    job.progress_percentage = 0
    job.progress_message = "Review submitted, waiting for the worker to apply changes"
    db.commit()                               # async: await db.commit(); await db.refresh(job)
    db.refresh(job)
    return job

from coder_agent.apply_review import apply_accepted

WAITING_FOR_REVIEW = "waiting_for_review"


def apply_review(job, report) -> None:
    """Phase 2: apply only the hunks the user accepted."""
    decision = job.review_decision or {"action": "none"}
    if not job.coder_run_id or not job.local_repo_path:
        raise ValueError("Job is missing coder_run_id or local_repo_path")

    label = {
        "all": "all changes",
        "none": "no changes (discarding)",
        "select": f"{len(decision.get('accepted', []))} selected hunk(s)",
    }.get(decision.get("action"), "changes")
    report(20, "review", f"Applying {label}")

    result = apply_accepted(
        run_id=job.coder_run_id,
        repo_path=job.local_repo_path,
        decision=decision,
    )

    message = result if isinstance(result, str) else getattr(result, "message", None)
    update_job(
        job.id,
        status=COMPLETED,
        progress_percentage=100,
        progress_message=(message or f"Applied {label}")[:2000],
    )
