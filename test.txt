import React, { useEffect, useMemo, useState } from "react";
import { useNavigate, useParams } from "react-router-dom";
import ReactMarkdown from "react-markdown";
import remarkGfm from "remark-gfm";
// Keep your existing import paths for these two:
import MainLayout from "../layouts/MainLayout";
import WorkflowNav from "../components/WorkflowNav";
import { Job, jobState } from "../hooks/useJobPolling";
import "./AssessmentPage.css";

const API = import.meta.env.VITE_API_BASE_URL;

type FileEntry = {
  path: string;
  description?: string;
  role_in_repository?: string;
  key_behavior?: string[];
  dependency_usage?: unknown[];
  important_symbols?: { name: string; kind?: string; purpose?: string }[];
  uncertainties?: unknown[];
};

type Tab = "skill" | "files";

async function getJson<T>(url: string): Promise<T> {
  const res = await fetch(url);
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body?.detail ?? `Request failed (${res.status})`);
  }
  return res.json();
}

export const AssessmentPage: React.FC = () => {
  const navigate = useNavigate();
  const { migrationId } = useParams();

  const [job, setJob] = useState<Job | null>(null);
  const [skillMd, setSkillMd] = useState<string | null>(null);
  const [files, setFiles] = useState<FileEntry[] | null>(null);
  const [error, setError] = useState<string | null>(null);

  const [tab, setTab] = useState<Tab>("skill");
  const [raw, setRaw] = useState(false);
  const [filter, setFilter] = useState("");
  const [reviewed, setReviewed] = useState(false);

  // Job + skill on page load
  useEffect(() => {
    if (!migrationId) return;
    let cancelled = false;

    (async () => {
      try {
        const j = await getJson<Job>(`${API}/jobs/${migrationId}`);
        if (cancelled) return;
        setJob(j);
        if (jobState(j.status) !== "done") return;

        const res = await fetch(`${API}/jobs/${migrationId}/artifacts/skill`);
        if (!res.ok) throw new Error("Could not load the generated skill");
        const text = await res.text();
        if (!cancelled) setSkillMd(text);
      } catch (e) {
        if (!cancelled) setError(e instanceof Error ? e.message : "Something went wrong");
      }
    })();

    return () => {
      cancelled = true;
    };
  }, [migrationId]);

  // File analysis only when its tab is opened
  useEffect(() => {
    if (tab !== "files" || files || !migrationId) return;
    getJson<Record<string, FileEntry>>(`${API}/jobs/${migrationId}/artifacts/file-analysis`)
      .then((data) => setFiles(Object.values(data)))
      .catch(() => setFiles([]));
  }, [tab, files, migrationId]);

  const visibleFiles = useMemo(() => {
    const q = filter.trim().toLowerCase();
    if (!files) return [];
    return q
      ? files.filter(
          (f) =>
            f.path.toLowerCase().includes(q) ||
            f.description?.toLowerCase().includes(q)
        )
      : files;
  }, [files, filter]);

  const downloadSkill = () => {
    if (!skillMd) return;
    const url = URL.createObjectURL(new Blob([skillMd], { type: "text/markdown" }));
    const a = document.createElement("a");
    a.href = url;
    a.download = `${job?.project_name ?? "repository"}_skill.md`;
    a.click();
    URL.revokeObjectURL(url);
  };

  const state = jobState(job?.status);

  return (
    <MainLayout
      title="Migration Skill Factory"
      subtitle="Review the generated skill"
      stepNumber={3}
      totalSteps={5}
      workflow={<WorkflowNav currentStep="assessment" migrationId={migrationId} />}
    >
      <div className="as-wrap">
        {error && <div className="as-alert">{error}</div>}
        {!job && !error && <div className="as-empty">Loading…</div>}

        {job && state !== "done" && (
          <div className="as-empty">
            This job is still <strong>{job.status}</strong>. Come back when the scan has finished.
            <button className="as-btn is-secondary" onClick={() => navigate(`/migration/${job.id}/scan`)}>
              Back to scan
            </button>
          </div>
        )}

        {job && state === "done" && (
          <>
            {/* ---------- Summary ---------- */}
            <section className="as-summary">
              <div>
                <span className="as-eyebrow">Assessment</span>
                <h2>{job.project_name}</h2>
                <p className="as-sub mono">
                  {job.repository_url} @ {job.branch || "main"}
                </p>
              </div>
              <div className="as-stats">
                <div>
                  <strong>{files ? files.length : "—"}</strong>
                  <span>files analysed</span>
                </div>
                <div>
                  <strong>{skillMd ? skillMd.split("\n").length : "—"}</strong>
                  <span>lines in skill</span>
                </div>
              </div>
            </section>

            {/* ---------- Tabs ---------- */}
            <section className="as-card">
              <div className="as-tabs" role="tablist">
                <button role="tab" aria-selected={tab === "skill"} className={tab === "skill" ? "is-active" : ""} onClick={() => setTab("skill")}>
                  Generated skill
                </button>
                <button role="tab" aria-selected={tab === "files"} className={tab === "files" ? "is-active" : ""} onClick={() => setTab("files")}>
                  File analysis
                </button>

                {tab === "skill" && (
                  <div className="as-tab-tools">
                    <button className="as-link" onClick={() => setRaw((r) => !r)}>
                      {raw ? "Rendered" : "Raw markdown"}
                    </button>
                    <button className="as-link" onClick={downloadSkill} disabled={!skillMd}>
                      Download .md
                    </button>
                  </div>
                )}
                {tab === "files" && (
                  <input
                    className="as-filter"
                    placeholder="Filter files…"
                    value={filter}
                    onChange={(e) => setFilter(e.target.value)}
                  />
                )}
              </div>

              <div className="as-body">
                {tab === "skill" &&
                  (skillMd === null ? (
                    <div className="as-empty">Loading skill…</div>
                  ) : raw ? (
                    <pre className="as-raw">{skillMd}</pre>
                  ) : (
                    <article className="as-md">
                      <ReactMarkdown remarkPlugins={[remarkGfm]}>{skillMd}</ReactMarkdown>
                    </article>
                  ))}

                {tab === "files" &&
                  (files === null ? (
                    <div className="as-empty">Loading file analysis…</div>
                  ) : visibleFiles.length === 0 ? (
                    <div className="as-empty">No files match.</div>
                  ) : (
                    <ul className="as-files">
                      {visibleFiles.map((f) => (
                        <li key={f.path}>
                          <details>
                            <summary>
                              <span className="mono as-path">{f.path}</span>
                              {f.role_in_repository && <span className="as-role">{f.role_in_repository}</span>}
                            </summary>
                            {f.description && <p>{f.description}</p>}
                            {!!f.key_behavior?.length && (
                              <ul className="as-bullets">
                                {f.key_behavior.map((b, i) => <li key={i}>{b}</li>)}
                              </ul>
                            )}
                            {!!f.important_symbols?.length && (
                              <div className="as-symbols">
                                {f.important_symbols.map((s, i) => (
                                  <span key={i} title={s.purpose} className="as-chip">
                                    <span className="mono">{s.name}</span>
                                    {s.kind && <em>{s.kind}</em>}
                                  </span>
                                ))}
                              </div>
                            )}
                          </details>
                        </li>
                      ))}
                    </ul>
                  ))}
              </div>
            </section>

            {/* ---------- Proceed ---------- */}
            <section className="as-footer">
              <label className="as-check">
                <input type="checkbox" checked={reviewed} onChange={(e) => setReviewed(e.target.checked)} />
                I have reviewed the generated skill and it looks correct
              </label>
              <button
                className="as-btn"
                disabled={!reviewed || !skillMd}
                onClick={() => navigate(`/migration/${job.id}/approval`)}
              >
                Proceed to Approval →
              </button>
            </section>
          </>
        )}
      </div>
    </MainLayout>
  );
};

export default AssessmentPage;

import uuid
from pathlib import Path

from fastapi import Depends, HTTPException
from fastapi.responses import FileResponse
from sqlalchemy.orm import Session

# from app.db.session import get_db      # <- use your existing imports
# from app.models.job import Job

ARTIFACTS: dict[str, tuple[str, str]] = {
    "skill": ("repository_skill.md", "text/markdown; charset=utf-8"),
    "file-analysis": ("file_analysis.json", "application/json"),
    "reading-plan": ("reading_plan.json", "application/json"),
}


@router.get("/{job_id}/artifacts/{name}")
def get_job_artifact(job_id: uuid.UUID, name: str, db: Session = Depends(get_db)):
    if name not in ARTIFACTS:
        raise HTTPException(404, f"Unknown artifact '{name}'. Use one of: {', '.join(ARTIFACTS)}")

    job = db.get(Job, job_id)          # async session: job = await db.get(Job, job_id)
    if job is None:
        raise HTTPException(404, "Job not found")
    if not job.output_path:
        raise HTTPException(409, "This job has no output yet")

    filename, media_type = ARTIFACTS[name]
    path = Path(job.output_path) / filename
    if not path.is_file():
        raise HTTPException(404, f"{filename} was not generated for this job")

    return FileResponse(path, media_type=media_type)
