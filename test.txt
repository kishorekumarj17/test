"""
Background job worker for Migration Skill Factory.

Every POLL_SECONDS:
  1. claims the oldest queued job (status -> running)
  2. pulls the repo if the job has no local_repo_path yet
  3. runs the main orchestration agent with the job details
  4. agents report progress through report(pct, [stage,] message) -> job row
  5. marks the job completed / failed

Run from the backend/ folder:
    python -m job_worker.worker
"""
from __future__ import annotations

import logging
import os
import signal
import time
import traceback
import uuid
from datetime import datetime
from pathlib import Path
from typing import Any, Callable, Optional

from dotenv import load_dotenv
from sqlalchemy import DateTime, Integer, String, Text, create_engine, func, select, update
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, sessionmaker

from job_worker.git_fetch import fetch_repo
from orchestrator.run import RunRequest, run_main_agent

load_dotenv()

# ---------------------------------------------------------------- config
# Must point at the SAME database the FastAPI app writes jobs to
# (migration_skill_db). Use a sync driver URL, e.g. postgresql+psycopg2://...
DB_URL = os.getenv("JOBS_DATABASE_URL") or os.environ["DATABASE_URL"]
POLL_SECONDS = int(os.getenv("WORKER_POLL_SECONDS", "5"))

QUEUED_STATUSES = ("queued", "pending")
RUNNING, COMPLETED, FAILED = "running", "completed", "failed"
SUCCESS_AGENT_STATUSES = {"completed", "success", "succeeded", "ok", "done"}

log = logging.getLogger("job_worker")
_stop = False

ReportFn = Callable[..., None]


# ---------------------------------------------------------------- model
# Mirror of app/app/models/job.py (the API owns the table and its Alembic
# migrations). Only map what the worker reads or writes. Never call
# Base.metadata.create_all() here.
class Base(DeclarativeBase):
    pass


class Job(Base):
    __tablename__ = "jobs"

    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True)
    project_name: Mapped[str] = mapped_column(String(255))
    repository_url: Mapped[str] = mapped_column(String(500))
    branch: Mapped[str | None] = mapped_column(String(500))
    local_repo_path: Mapped[str | None] = mapped_column(String(500))
    output_path: Mapped[str | None] = mapped_column(String(500))
    description: Mapped[str | None] = mapped_column(Text)
    progress_percentage: Mapped[int] = mapped_column(Integer, default=0)
    status: Mapped[str] = mapped_column(String(50))
    job_type: Mapped[str | None] = mapped_column(String(50))
    user_query: Mapped[str | None] = mapped_column(Text)
    progress_message: Mapped[str | None] = mapped_column(Text)
    created_at: Mapped[datetime] = mapped_column(DateTime, server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(
        DateTime, server_default=func.now(), onupdate=func.now()
    )


# ---------------------------------------------------------------- db helpers
engine = create_engine(
    DB_URL.replace("+asyncpg", "+psycopg2"),  # worker is sync
    pool_pre_ping=True,
    pool_size=2,
    max_overflow=2,
)
# expire_on_commit=False: the claimed Job stays readable after the session closes
SessionLocal = sessionmaker(bind=engine, expire_on_commit=False)


def claim_next_job() -> Optional[Job]:
    """Atomically take the oldest queued job and mark it running.
    SKIP LOCKED makes it safe to run more than one worker."""
    with SessionLocal.begin() as session:
        job = session.scalars(
            select(Job)
            .where(Job.status.in_(QUEUED_STATUSES))
            .order_by(Job.created_at.asc())
            .limit(1)
            .with_for_update(skip_locked=True)
        ).first()
        if job is None:
            return None
        job.status = RUNNING
        job.progress_percentage = 0
        job.progress_message = "Picked up by worker"
        return job  # committed when the block exits


def update_job(job_id: uuid.UUID, **fields: Any) -> None:
    """Short transaction per update so the UI sees progress immediately."""
    with SessionLocal.begin() as session:
        session.execute(update(Job).where(Job.id == job_id).values(**fields))


# ---------------------------------------------------------------- progress
def make_reporter(job_id: uuid.UUID) -> ReportFn:
    """Returns report(percent, message) or report(percent, stage, message).

    - percent is clamped to 0..100 and never goes backwards
    - failures to write progress are logged, never raised into the agent
    """
    last = {"pct": 0}

    def report(percent: int, stage_or_message: str, message: Optional[str] = None) -> None:
        stage, msg = (stage_or_message, message) if message is not None else (None, stage_or_message)
        pct = max(last["pct"], min(100, max(0, int(percent))))
        last["pct"] = pct
        log.info("job %s | %3d%% | %s%s", job_id, pct, f"[{stage}] " if stage else "", msg)
        try:
            update_job(job_id, progress_percentage=pct, progress_message=str(msg)[:2000])
        except Exception:  # never break the agent because of a progress write
            log.exception("could not write progress for job %s", job_id)

    return report


# ---------------------------------------------------------------- job processing
def build_query(job: Job, repo_path: Path) -> str:
    query = f"create skills using this path {repo_path}"
    if job.user_query:
        query += f". {job.user_query}"
    return query


def process_job(job: Job) -> None:
    job_id = job.id
    report = make_reporter(job_id)
    log.info("job %s claimed: %s", job_id, job.project_name)

    try:
        # 1. source code
        local = job.local_repo_path
        if local and Path(local).exists():
            repo_path = Path(local)
            report(10, "git", f"Using existing local repository {repo_path}")
        else:
            repo_url = job.repository_url
            branch = job.branch
            if branch:
                repo_url = f"{repo_url}#{branch}"  # git_fetch understands repo#branch
            repo_path = fetch_repo(job.project_name, repo_url, report)
            update_job(job_id, local_repo_path=str(repo_path))

        # 2. main agent
        report(25, "agent", "Starting main agent")
        request = RunRequest(query=build_query(job, repo_path), selected_skill_ids=[])
        response = run_main_agent(request, report=report)

        # 3. result
        agent_status = str(getattr(response, "status", "")).lower()
        output_path = getattr(response, "output_path", None) or getattr(response, "output_dir", None)

        if agent_status in SUCCESS_AGENT_STATUSES:
            update_job(
                job_id,
                status=COMPLETED,
                progress_percentage=100,
                progress_message="Completed",
                output_path=str(output_path) if output_path else None,
            )
            log.info("job %s completed", job_id)
        else:
            reason = getattr(response, "message", None) or getattr(response, "error", None) or agent_status
            update_job(job_id, status=FAILED, progress_message=f"Agent stopped: {reason}"[:2000])
            log.warning("job %s failed: %s", job_id, reason)

    except Exception as exc:
        log.error("job %s crashed:\n%s", job_id, traceback.format_exc())
        update_job(job_id, status=FAILED, progress_message=f"Error: {exc}"[:2000])


# ---------------------------------------------------------------- main loop
def _handle_stop(*_: Any) -> None:
    global _stop
    _stop = True
    log.info("stop requested, finishing current job…")


def main() -> None:
    logging.basicConfig(
        level=os.getenv("LOG_LEVEL", "INFO"),
        format="%(asctime)s | %(levelname)-7s | %(name)s | %(message)s",
    )
    signal.signal(signal.SIGINT, _handle_stop)
    signal.signal(signal.SIGTERM, _handle_stop)

    log.info("worker started, polling jobs every %ss", POLL_SECONDS)

    while not _stop:
        try:
            job = claim_next_job()
        except Exception:
            log.exception("could not poll for jobs")
            job = None

        if job:
            process_job(job)  # check again immediately after a job
            continue

        for _ in range(POLL_SECONDS * 10):  # sleep, but react quickly to Ctrl+C
            if _stop:
                break
            time.sleep(0.1)

    log.info("worker stopped")


if __name__ == "__main__":
    main()
