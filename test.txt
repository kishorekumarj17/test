/**
 * Minimal unified-diff parser.
 *
 * Hunk IDs are "<fileIndex>:<hunkIndex>", both 0-based in the order they
 * appear in full.diff. This MUST match the IDs produced by list_hunks()
 * in coder_agent/apply_review.py — verify once against a real run.
 */

export type DiffLine = {
  type: "add" | "del" | "ctx" | "meta";
  text: string;
  oldNo?: number;
  newNo?: number;
};

export type Hunk = {
  id: string;
  fileIndex: number;
  hunkIndex: number;
  header: string;
  lines: DiffLine[];
  adds: number;
  dels: number;
};

export type FileDiff = {
  index: number;
  path: string;
  oldPath: string | null;
  newPath: string | null;
  isNew: boolean;
  isDeleted: boolean;
  isBinary: boolean;
  hunks: Hunk[];
  adds: number;
  dels: number;
};

const HUNK_RE = /^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@/;
const stripPrefix = (p: string) =>
  p === "/dev/null" ? null : p.replace(/^[ab]\//, "").replace(/\t.*$/, "");

export function parseDiff(text: string): FileDiff[] {
  const files: FileDiff[] = [];
  let file: FileDiff | null = null;
  let hunk: Hunk | null = null;
  let oldLeft = 0;
  let newLeft = 0;
  let oldNo = 0;
  let newNo = 0;

  const startFile = (path = "") => {
    file = {
      index: files.length,
      path,
      oldPath: null,
      newPath: null,
      isNew: false,
      isDeleted: false,
      isBinary: false,
      hunks: [],
      adds: 0,
      dels: 0,
    };
    files.push(file);
    hunk = null;
  };

  for (const raw of text.replace(/\r\n/g, "\n").split("\n")) {
    // Inside a hunk: consume exactly the number of lines the header promised
    if (hunk && file && (oldLeft > 0 || newLeft > 0)) {
      const c = raw[0];
      if (c === "+") {
        hunk.lines.push({ type: "add", text: raw.slice(1), newNo: newNo++ });
        hunk.adds++; file.adds++; newLeft--;
        continue;
      }
      if (c === "-") {
        hunk.lines.push({ type: "del", text: raw.slice(1), oldNo: oldNo++ });
        hunk.dels++; file.dels++; oldLeft--;
        continue;
      }
      if (c === " " || raw === "") {
        hunk.lines.push({ type: "ctx", text: raw.slice(1), oldNo: oldNo++, newNo: newNo++ });
        oldLeft--; newLeft--;
        continue;
      }
      if (c === "\\") {
        hunk.lines.push({ type: "meta", text: raw });
        continue;
      }
      // malformed hunk: fall through and treat as a header line
      oldLeft = newLeft = 0;
    }
    if (raw.startsWith("\\") && hunk) {
      hunk.lines.push({ type: "meta", text: raw });
      continue;
    }

    if (raw.startsWith("diff --git ")) {
      const m = raw.match(/^diff --git a\/(.+?) b\/(.+)$/);
      startFile(m ? m[2] : raw.slice(11));
      continue;
    }

    if (raw.startsWith("--- ")) {
      // plain unified diff without "diff --git" headers
      if (!file || (file as FileDiff).hunks.length > 0) startFile();
      file!.oldPath = stripPrefix(raw.slice(4));
      continue;
    }
    if (raw.startsWith("+++ ") && file) {
      file.newPath = stripPrefix(raw.slice(4));
      file.path = file.newPath ?? file.oldPath ?? file.path;
      file.isNew = file.oldPath === null && !!file.newPath;
      file.isDeleted = file.newPath === null;
      continue;
    }

    if (!file) continue;
    if (raw.startsWith("new file mode")) file.isNew = true;
    else if (raw.startsWith("deleted file mode")) file.isDeleted = true;
    else if (raw.startsWith("Binary files")) file.isBinary = true;
    else {
      const m = raw.match(HUNK_RE);
      if (m) {
        const hunkIndex = file.hunks.length;
        hunk = {
          id: `${file.index}:${hunkIndex}`,
          fileIndex: file.index,
          hunkIndex,
          header: raw,
          lines: [],
          adds: 0,
          dels: 0,
        };
        file.hunks.push(hunk);
        oldNo = +m[1];
        newNo = +m[3];
        oldLeft = m[2] === undefined ? 1 : +m[2];
        newLeft = m[4] === undefined ? 1 : +m[4];
      }
    }
  }

  return files;
}


import { useCallback, useEffect, useRef, useState } from "react";

export type Job = {
  id: string;
  project_name: string;
  repository_url: string;
  branch?: string | null;
  description?: string | null;
  progress_percentage: number | null;
  status: string;
  job_type?: string | null;
  user_query?: string | null;
  progress_message?: string | null;
  local_repo_path?: string | null;
  output_path?: string | null;
  coder_run_id?: string | null;
  review_decision?: { action: "all" | "none" | "select"; accepted?: string[] } | null;
  created_at: string;
  updated_at: string;
};

const DONE = ["completed", "complete", "success", "succeeded", "done"];
const FAILED = ["failed", "error", "cancelled", "canceled"];
const REVIEW = ["waiting_for_review", "review"];

export const jobState = (status?: string) => {
  const s = (status ?? "").toLowerCase();
  if (DONE.includes(s)) return "done" as const;
  if (FAILED.includes(s)) return "failed" as const;
  if (REVIEW.includes(s)) return "review" as const;
  if (["pending", "queued", "created", "new"].includes(s)) return "queued" as const;
  return "running" as const;
};

/**
 * Fetches GET /jobs/{id} every `intervalMs` until the job completes, fails or
 * waits for review. Change `pollKey` to start polling again (e.g. after
 * submitting a review, when the job goes back to "queued").
 */
export function useJobPolling(jobId?: string, intervalMs = 4000, pollKey: unknown = 0) {
  const [job, setJob] = useState<Job | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [history, setHistory] = useState<{ at: string; message: string }[]>([]);
  const timer = useRef<number | null>(null);

  const fetchJob = useCallback(async () => {
    if (!jobId) return null;
    try {
      // Adjust the path if your API uses a different one
      const res = await fetch(`${import.meta.env.VITE_API_BASE_URL}/jobs/${jobId}`);
      if (!res.ok) throw new Error(`Could not load job (${res.status})`);
      const data: Job = await res.json();
      setJob(data);
      setError(null);
      if (data.progress_message) {
        setHistory((h) =>
          h.length && h[h.length - 1].message === data.progress_message
            ? h
            : [...h, { at: data.updated_at, message: data.progress_message! }]
        );
      }
      return data;
    } catch (e) {
      setError(e instanceof Error ? e.message : "Could not load job");
      return null;
    }
  }, [jobId]);

  // Clear the previous job's data when a different job is tracked
  useEffect(() => {
    setJob(null);
    setHistory([]);
  }, [jobId]);

  useEffect(() => {
    if (!jobId) return; // nothing to track yet
    let cancelled = false;

    const tick = async () => {
      const data = await fetchJob();
      if (cancelled) return;
      const state = jobState(data?.status);
      if (state === "done" || state === "failed" || state === "review") return; // stop polling
      timer.current = window.setTimeout(tick, intervalMs);
    };

    tick();
    return () => {
      cancelled = true;
      if (timer.current) window.clearTimeout(timer.current);
    };
  }, [fetchJob, intervalMs, pollKey]);

  return { job, error, history, refresh: fetchJob };
}
