import React, { useEffect, useState } from "react";
import { useNavigate, useParams } from "react-router-dom";
// Keep your existing import paths for these two:
import MainLayout from "../layouts/MainLayout";
import WorkflowNav from "../components/WorkflowNav";
import { Job, jobState } from "../hooks/useJobPolling";
import "./ApprovalPage.css";

const API = import.meta.env.VITE_API_BASE_URL;
const MAX = 2000;

const EXAMPLES = [
  "Upgrade React from 17 to 18 and fix breaking changes",
  "Replace moment.js with date-fns",
  "Migrate class components to function components with hooks",
  "Upgrade all internal @org packages to their latest major version",
];

type JobWithPaths = Job & {
  local_repo_path?: string | null;
  output_path?: string | null;
};

export const ApprovalPage: React.FC = () => {
  const navigate = useNavigate();
  const { migrationId } = useParams();

  const [job, setJob] = useState<JobWithPaths | null>(null);
  const [query, setQuery] = useState("");
  const [loadError, setLoadError] = useState<string | null>(null);
  const [submitError, setSubmitError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  useEffect(() => {
    if (!migrationId) return;
    let cancelled = false;
    fetch(`${API}/jobs/${migrationId}`)
      .then((r) => (r.ok ? r.json() : Promise.reject(new Error(`Could not load job (${r.status})`))))
      .then((j) => !cancelled && setJob(j))
      .catch((e) => !cancelled && setLoadError(e.message));
    return () => {
      cancelled = true;
    };
  }, [migrationId]);

  const ready = !!job && jobState(job.status) === "done" && !!job.local_repo_path;
  const canSubmit = ready && query.trim().length >= 10 && !submitting;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!job || !canSubmit) return;
    setSubmitting(true);
    setSubmitError(null);

    try {
      // New job = copy of the skill job + the change request, typed as "code"
      const res = await fetch(`${API}/jobs`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          project_name: job.project_name,
          repository_url: job.repository_url,
          branch: job.branch || "main",
          description: job.description ?? null,
          local_repo_path: job.local_repo_path,
          output_path: job.output_path ?? null,
          job_type: "code",
          user_query: query.trim(),
        }),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(
          typeof body?.detail === "string" ? body.detail : `Could not create the job (${res.status})`
        );
      }
      const created: Job = await res.json();
      navigate(`/migration/${created.id}/upgrade`);
    } catch (err) {
      setSubmitError(err instanceof Error ? err.message : "Something went wrong");
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <MainLayout
      title="Migration Skill Factory"
      subtitle="Describe the code change"
      stepNumber={4}
      totalSteps={5}
      workflow={<WorkflowNav currentStep="approval" migrationId={migrationId} disabled={submitting} />}
    >
      <div className="ap-wrap">
        {loadError && <div className="ap-alert">{loadError}</div>}
        {!job && !loadError && <div className="ap-note">Loading…</div>}

        {job && !ready && (
          <div className="ap-note">
            {jobState(job.status) !== "done"
              ? "The skill job hasn't finished yet."
              : "This job has no local repository path, so the coder agent has nothing to work on."}
            <button className="ap-btn is-secondary" onClick={() => navigate(`/migration/${job.id}/scan`)}>
              Back to scan
            </button>
          </div>
        )}

        {job && ready && (
          <section className="ap-card">
            {/* Left: what the coder agent will use */}
            <aside className="ap-context">
              <span className="ap-eyebrow">Step 4 · Approval</span>
              <h2>{job.project_name}</h2>
              <p className="ap-muted">
                The coder agent applies your change to this repository, guided by the skill you just
                reviewed, and produces a diff for you to validate.
              </p>

              <dl className="ap-meta">
                <div>
                  <dt>Repository</dt>
                  <dd className="mono" title={job.repository_url}>{job.repository_url}</dd>
                </div>
                <div>
                  <dt>Branch</dt>
                  <dd className="mono">{job.branch || "main"}</dd>
                </div>
                <div>
                  <dt>Working copy</dt>
                  <dd className="mono" title={job.local_repo_path ?? ""}>{job.local_repo_path}</dd>
                </div>
                <div>
                  <dt>Skill</dt>
                  <dd className="ap-ok">✓ Reviewed</dd>
                </div>
              </dl>
            </aside>

            {/* Right: the change request */}
            <form className="ap-form" onSubmit={handleSubmit}>
              <label htmlFor="query" className="ap-label">
                What should the coder agent change? <span className="ap-req">*</span>
              </label>
              <textarea
                id="query"
                className="ap-textarea"
                placeholder="e.g. Upgrade the app from React 17 to 18. Replace ReactDOM.render with createRoot and fix any type errors."
                value={query}
                onChange={(e) => setQuery(e.target.value)}
                maxLength={MAX}
                rows={8}
                disabled={submitting}
              />
              <div className="ap-row">
                <span className="ap-hint">Be specific: versions, packages, files or patterns to change.</span>
                <span className="ap-hint">{query.length}/{MAX}</span>
              </div>

              <div className="ap-examples">
                <span className="ap-hint">Examples</span>
                <div>
                  {EXAMPLES.map((ex) => (
                    <button key={ex} type="button" className="ap-chip" onClick={() => setQuery(ex)} disabled={submitting}>
                      {ex}
                    </button>
                  ))}
                </div>
              </div>

              {submitError && <div className="ap-alert">{submitError}</div>}

              <div className="ap-actions">
                <button type="button" className="ap-btn is-secondary" onClick={() => navigate(`/migration/${job.id}/assessment`)} disabled={submitting}>
                  ← Back
                </button>
                <button type="submit" className="ap-btn" disabled={!canSubmit}>
                  {submitting ? "Submitting…" : "Submit code change →"}
                </button>
              </div>
            </form>
          </section>
        )}
      </div>
    </MainLayout>
  );
};

export default ApprovalPage;

from pathlib import Path

CODE_JOB_TYPES = {"code"}

# Static instruction that makes the planner pick the coder agent only
CODER_INSTRUCTION = (
    "Run only the coder_agent. Do not generate or update skills. "
    "Apply the change request below to the repository and produce a diff."
)


def is_code_job(job) -> bool:
    return (job.job_type or "").lower() in CODE_JOB_TYPES


def build_query(job, repo_path: Path) -> str:
    if is_code_job(job):
        lines = [CODER_INSTRUCTION, f"Repository path: {repo_path}"]
        if job.output_path:
            lines.append(f"Use the repository skill at: {Path(job.output_path) / 'repository_skill.md'}")
        lines.append(f"Change request: {job.user_query}")
        return "\n".join(lines)

    # skill job (unchanged)
    query = f"create skills using this path {repo_path}"
    if job.user_query:
        query += f". {job.user_query}"
    return query
