import React from "react";
import { useNavigate, useParams } from "react-router-dom";
// Keep your existing import paths for these two:
import MainLayout from "../layouts/MainLayout";
import WorkflowNav from "../components/WorkflowNav";
import { useJobPolling, jobState } from "../hooks/useJobPolling";
import "./ScanPage.css";

const STATE_LABEL = {
  queued: "Queued",
  running: "Scanning",
  done: "Completed",
  failed: "Failed",
} as const;

const formatTime = (iso?: string) =>
  iso
    ? new Date(iso).toLocaleString(undefined, {
        day: "2-digit",
        month: "short",
        hour: "2-digit",
        minute: "2-digit",
      })
    : "—";

const repoLabel = (url?: string) =>
  url?.replace(/^https?:\/\/|^git@/, "").replace(/\.git$/, "").replace(":", "/") ?? "—";

export const ScanPage: React.FC = () => {
  const navigate = useNavigate();
  const { migrationId } = useParams();
  const { job, error, history } = useJobPolling(migrationId, 4000);

  const state = jobState(job?.status);
  const pct = Math.max(0, Math.min(100, job?.progress_percentage ?? 0));
  const isActive = state === "queued" || state === "running";

  return (
    <MainLayout
      title="Migration Skill Factory"
      subtitle="Scanning repository"
      stepNumber={2}
      totalSteps={5}
      workflow={
        <WorkflowNav
          currentStep="scan"
          migrationId={migrationId}
          disabled={isActive}
        />
      }
    >
      <div className="scan-wrap">
        {error && !job && (
          <div className="scan-alert" role="alert">
            {error}
          </div>
        )}

        {!job && !error && <div className="scan-skeleton">Loading job…</div>}

        {job && (
          <>
            {/* ---------- Job details ---------- */}
            <section className="scan-card scan-details">
              <div className="scan-details-head">
                <div>
                  <span className="scan-eyebrow">Application</span>
                  <h2>{job.project_name}</h2>
                </div>
                <span className={`scan-badge is-${state}`}>
                  {isActive && <span className="scan-dot" />}
                  {STATE_LABEL[state]}
                </span>
              </div>

              {job.description && <p className="scan-desc">{job.description}</p>}

              <dl className="scan-meta">
                <div>
                  <dt>Repository</dt>
                  <dd className="mono" title={job.repository_url}>
                    {repoLabel(job.repository_url)}
                  </dd>
                </div>
                <div>
                  <dt>Branch</dt>
                  <dd className="mono">{job.branch || "main"}</dd>
                </div>
                <div>
                  <dt>Job type</dt>
                  <dd>{job.job_type || "—"}</dd>
                </div>
                <div>
                  <dt>Started</dt>
                  <dd>{formatTime(job.created_at)}</dd>
                </div>
                <div>
                  <dt>Job ID</dt>
                  <dd className="mono small">{job.id}</dd>
                </div>
              </dl>
            </section>

            {/* ---------- Progress ---------- */}
            <section className="scan-card scan-progress">
              <div className="scan-progress-top">
                <h3>Progress</h3>
                <span className="scan-pct">{pct}%</span>
              </div>

              <div
                className={`scan-bar is-${state}`}
                role="progressbar"
                aria-valuemin={0}
                aria-valuemax={100}
                aria-valuenow={pct}
              >
                <div className="scan-bar-fill" style={{ width: `${pct}%` }} />
              </div>

              <p className={`scan-message is-${state}`} aria-live="polite">
                {job.progress_message ||
                  (state === "queued"
                    ? "Waiting for a worker to pick up this job…"
                    : "Working…")}
              </p>

              {history.length > 1 && (
                <ol className="scan-log">
                  {history
                    .slice()
                    .reverse()
                    .map((h, i) => (
                      <li key={`${h.at}-${i}`} className={i === 0 ? "is-latest" : ""}>
                        <span className="scan-log-time">{formatTime(h.at)}</span>
                        <span>{h.message}</span>
                      </li>
                    ))}
                </ol>
              )}

              <div className="scan-footer">
                <span className="scan-updated">
                  {isActive
                    ? "Refreshing every 4 seconds"
                    : `Last updated ${formatTime(job.updated_at)}`}
                </span>

                {state === "done" && (
                  <button
                    className="scan-btn"
                    onClick={() => navigate(`/migration/${job.id}/assessment`)}
                  >
                    Continue to Assessment →
                  </button>
                )}

                {state === "failed" && (
                  <button
                    className="scan-btn is-secondary"
                    onClick={() => navigate("/")}
                  >
                    Start a new scan
                  </button>
                )}
              </div>
            </section>
          </>
        )}
      </div>
    </MainLayout>
  );
};

export default ScanPage;

import { useCallback, useEffect, useRef, useState } from "react";

export type Job = {
  id: string;
  project_name: string;
  repository_url: string;
  branch?: string | null;
  description?: string | null;
  progress_percentage: number | null;
  status: string;
  job_type?: string | null;
  user_query?: string | null;
  progress_message?: string | null;
  created_at: string;
  updated_at: string;
};

const DONE = ["completed", "complete", "success", "succeeded", "done"];
const FAILED = ["failed", "error", "cancelled", "canceled"];

export const jobState = (status?: string) => {
  const s = (status ?? "").toLowerCase();
  if (DONE.includes(s)) return "done" as const;
  if (FAILED.includes(s)) return "failed" as const;
  if (["pending", "queued", "created", "new"].includes(s)) return "queued" as const;
  return "running" as const;
};

/** Fetches GET /jobs/{id} every `intervalMs` until the job completes or fails. */
export function useJobPolling(jobId?: string, intervalMs = 4000) {
  const [job, setJob] = useState<Job | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [history, setHistory] = useState<{ at: string; message: string }[]>([]);
  const timer = useRef<number | null>(null);

  const fetchJob = useCallback(async () => {
    if (!jobId) return null;
    try {
      // Adjust the path if your API uses a different one
      const res = await fetch(`${import.meta.env.VITE_API_BASE_URL}/jobs/${jobId}`);
      if (!res.ok) throw new Error(`Could not load job (${res.status})`);
      const data: Job = await res.json();
      setJob(data);
      setError(null);
      if (data.progress_message) {
        setHistory((h) =>
          h.length && h[h.length - 1].message === data.progress_message
            ? h
            : [...h, { at: data.updated_at, message: data.progress_message! }]
        );
      }
      return data;
    } catch (e) {
      setError(e instanceof Error ? e.message : "Could not load job");
      return null;
    }
  }, [jobId]);

  useEffect(() => {
    let cancelled = false;

    const tick = async () => {
      const data = await fetchJob();
      if (cancelled) return;
      const state = jobState(data?.status);
      if (state === "done" || state === "failed") return; // stop polling
      timer.current = window.setTimeout(tick, intervalMs);
    };

    tick();
    return () => {
      cancelled = true;
      if (timer.current) window.clearTimeout(timer.current);
    };
  }, [fetchJob, intervalMs]);

  return { job, error, history, refresh: fetchJob };
}
