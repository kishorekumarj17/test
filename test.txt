import React, { useEffect, useState } from "react";
import { useNavigate, useParams, useSearchParams } from "react-router-dom";
// Keep your existing import paths for these two:
import MainLayout from "../layouts/MainLayout";
import WorkflowNav from "../components/WorkflowNav";
import { Job, jobState, useJobPolling } from "../hooks/useJobPolling";
import "./ApprovalPage.css";

const API = import.meta.env.VITE_API_BASE_URL;
const MAX = 2000;

const EXAMPLES = [
  "Upgrade React from 17 to 18 and fix breaking changes",
  "Replace moment.js with date-fns",
  "Migrate class components to function components with hooks",
  "Upgrade all internal @org packages to their latest major version",
];

type JobWithPaths = Job & {
  local_repo_path?: string | null;
  output_path?: string | null;
};

export const ApprovalPage: React.FC = () => {
  const navigate = useNavigate();
  const { migrationId } = useParams();

  const [job, setJob] = useState<JobWithPaths | null>(null);
  const [query, setQuery] = useState("");
  const [loadError, setLoadError] = useState<string | null>(null);
  const [submitError, setSubmitError] = useState<string | null>(null);
  const [submitting, setSubmitting] = useState(false);

  // The code job being tracked lives in the URL (?codeJob=<id>) so a refresh keeps tracking it
  const [searchParams, setSearchParams] = useSearchParams();
  const codeJobId = searchParams.get("codeJob") ?? undefined;
  const { job: codeJob, error: codeJobError, history } = useJobPolling(codeJobId, 4000);
  const codeState = jobState(codeJob?.status);
  const codePct = Math.max(0, Math.min(100, codeJob?.progress_percentage ?? 0));
  const tracking = !!codeJobId;
  const codeRunning = tracking && (codeState === "queued" || codeState === "running");

  useEffect(() => {
    if (!migrationId) return;
    let cancelled = false;
    fetch(`${API}/jobs/${migrationId}`)
      .then((r) => (r.ok ? r.json() : Promise.reject(new Error(`Could not load job (${r.status})`))))
      .then((j) => !cancelled && setJob(j))
      .catch((e) => !cancelled && setLoadError(e.message));
    return () => {
      cancelled = true;
    };
  }, [migrationId]);

  const ready = !!job && jobState(job.status) === "done" && !!job.local_repo_path;
  const canSubmit = ready && !tracking && query.trim().length >= 10 && !submitting;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!job || !canSubmit) return;
    setSubmitting(true);
    setSubmitError(null);

    try {
      // New job = copy of the skill job + the change request, typed as "code"
      const res = await fetch(`${API}/jobs`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          project_name: job.project_name,
          repository_url: job.repository_url,
          branch: job.branch || "main",
          description: job.description ?? null,
          local_repo_path: job.local_repo_path,
          output_path: job.output_path ?? null,
          job_type: "code",
          user_query: query.trim(),
        }),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(
          typeof body?.detail === "string" ? body.detail : `Could not create the job (${res.status})`
        );
      }
      const created: Job = await res.json();
      setSearchParams({ codeJob: String(created.id) }); // stay here and track it
    } catch (err) {
      setSubmitError(err instanceof Error ? err.message : "Something went wrong");
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <MainLayout
      title="Migration Skill Factory"
      subtitle="Describe the code change"
      stepNumber={4}
      totalSteps={5}
      workflow={<WorkflowNav currentStep="approval" migrationId={migrationId} disabled={submitting || codeRunning} />}
    >
      <div className="ap-wrap">
        {loadError && <div className="ap-alert">{loadError}</div>}
        {!job && !loadError && <div className="ap-note">Loading…</div>}

        {job && !ready && (
          <div className="ap-note">
            {jobState(job.status) !== "done"
              ? "The skill job hasn't finished yet."
              : "This job has no local repository path, so the coder agent has nothing to work on."}
            <button className="ap-btn is-secondary" onClick={() => navigate(`/migration/${job.id}/scan`)}>
              Back to scan
            </button>
          </div>
        )}

        {job && ready && (
          <section className="ap-card">
            {/* Left: what the coder agent will use */}
            <aside className="ap-context">
              <span className="ap-eyebrow">Step 4 · Approval</span>
              <h2>{job.project_name}</h2>
              <p className="ap-muted">
                The coder agent applies your change to this repository, guided by the skill you just
                reviewed, and produces a diff for you to validate.
              </p>

              <dl className="ap-meta">
                <div>
                  <dt>Repository</dt>
                  <dd className="mono" title={job.repository_url}>{job.repository_url}</dd>
                </div>
                <div>
                  <dt>Branch</dt>
                  <dd className="mono">{job.branch || "main"}</dd>
                </div>
                <div>
                  <dt>Working copy</dt>
                  <dd className="mono" title={job.local_repo_path ?? ""}>{job.local_repo_path}</dd>
                </div>
                <div>
                  <dt>Skill</dt>
                  <dd className="ap-ok">✓ Reviewed</dd>
                </div>
              </dl>
            </aside>

            {/* Right: the change request */}
            <form className="ap-form" onSubmit={handleSubmit}>
              <label htmlFor="query" className="ap-label">
                What should the coder agent change? <span className="ap-req">*</span>
              </label>
              <textarea
                id="query"
                className="ap-textarea"
                placeholder="e.g. Upgrade the app from React 17 to 18. Replace ReactDOM.render with createRoot and fix any type errors."
                value={tracking ? codeJob?.user_query ?? query : query}
                onChange={(e) => setQuery(e.target.value)}
                maxLength={MAX}
                rows={8}
                disabled={submitting || tracking}
              />
              <div className="ap-row">
                <span className="ap-hint">Be specific: versions, packages, files or patterns to change.</span>
                <span className="ap-hint">{query.length}/{MAX}</span>
              </div>

              <div className="ap-examples">
                <span className="ap-hint">Examples</span>
                <div>
                  {EXAMPLES.map((ex) => (
                    <button key={ex} type="button" className="ap-chip" onClick={() => setQuery(ex)} disabled={submitting || tracking}>
                      {ex}
                    </button>
                  ))}
                </div>
              </div>

              {submitError && <div className="ap-alert">{submitError}</div>}

              {!tracking && (
                <div className="ap-actions">
                  <button type="button" className="ap-btn is-secondary" onClick={() => navigate(`/migration/${job.id}/assessment`)} disabled={submitting}>
                    ← Back
                  </button>
                  <button type="submit" className="ap-btn" disabled={!canSubmit}>
                    {submitting ? "Submitting…" : "Submit code change →"}
                  </button>
                </div>
              )}

              {/* ---------- Code job progress ---------- */}
              {tracking && (
                <div className={`ap-progress is-${codeState}`} aria-live="polite">
                  <div className="ap-progress-top">
                    <div>
                      <span className="ap-hint">Code job</span>
                      <div className="mono ap-jobid">{codeJobId}</div>
                    </div>
                    <span className={`ap-badge is-${codeState}`}>
                      {codeRunning && <span className="ap-dot" />}
                      {{ queued: "Queued", running: "Coding", done: "Completed", failed: "Failed" }[codeState]}
                    </span>
                  </div>

                  <div className="ap-bar-row">
                    <div className="ap-bar" role="progressbar" aria-valuemin={0} aria-valuemax={100} aria-valuenow={codePct}>
                      <div className="ap-bar-fill" style={{ width: `${codePct}%` }} />
                    </div>
                    <span className="ap-pct">{codePct}%</span>
                  </div>

                  <p className="ap-message">
                    {codeJobError && !codeJob
                      ? codeJobError
                      : codeJob?.progress_message ||
                        (codeState === "queued" ? "Waiting for the worker to pick up this job…" : "Working…")}
                  </p>

                  {history.length > 1 && (
                    <ol className="ap-log">
                      {history.slice(0, -1).reverse().slice(0, 5).map((h, i) => (
                        <li key={`${h.at}-${i}`}>{h.message}</li>
                      ))}
                    </ol>
                  )}

                  <div className="ap-actions">
                    {codeState === "failed" ? (
                      <button type="button" className="ap-btn is-secondary" onClick={() => setSearchParams({})}>
                        Edit request and retry
                      </button>
                    ) : (
                      <span className="ap-hint">{codeRunning ? "Refreshing every 4 seconds" : ""}</span>
                    )}
                    <button
                      type="button"
                      className="ap-btn"
                      disabled={codeState !== "done"}
                      onClick={() => navigate(`/migration/${codeJobId}/upgrade`)}
                    >
                      Continue to Validation →
                    </button>
                  </div>
                </div>
              )}
            </form>
          </section>
        )}
      </div>
    </MainLayout>
  );
};

export default ApprovalPage;

import { useCallback, useEffect, useRef, useState } from "react";

export type Job = {
  id: string;
  project_name: string;
  repository_url: string;
  branch?: string | null;
  description?: string | null;
  progress_percentage: number | null;
  status: string;
  job_type?: string | null;
  user_query?: string | null;
  progress_message?: string | null;
  created_at: string;
  updated_at: string;
};

const DONE = ["completed", "complete", "success", "succeeded", "done"];
const FAILED = ["failed", "error", "cancelled", "canceled"];

export const jobState = (status?: string) => {
  const s = (status ?? "").toLowerCase();
  if (DONE.includes(s)) return "done" as const;
  if (FAILED.includes(s)) return "failed" as const;
  if (["pending", "queued", "created", "new"].includes(s)) return "queued" as const;
  return "running" as const;
};

/** Fetches GET /jobs/{id} every `intervalMs` until the job completes or fails. */
export function useJobPolling(jobId?: string, intervalMs = 4000) {
  const [job, setJob] = useState<Job | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [history, setHistory] = useState<{ at: string; message: string }[]>([]);
  const timer = useRef<number | null>(null);

  const fetchJob = useCallback(async () => {
    if (!jobId) return null;
    try {
      // Adjust the path if your API uses a different one
      const res = await fetch(`${import.meta.env.VITE_API_BASE_URL}/jobs/${jobId}`);
      if (!res.ok) throw new Error(`Could not load job (${res.status})`);
      const data: Job = await res.json();
      setJob(data);
      setError(null);
      if (data.progress_message) {
        setHistory((h) =>
          h.length && h[h.length - 1].message === data.progress_message
            ? h
            : [...h, { at: data.updated_at, message: data.progress_message! }]
        );
      }
      return data;
    } catch (e) {
      setError(e instanceof Error ? e.message : "Could not load job");
      return null;
    }
  }, [jobId]);

  useEffect(() => {
    if (!jobId) return; // nothing to track yet
    let cancelled = false;
    setJob(null);
    setHistory([]);

    const tick = async () => {
      const data = await fetchJob();
      if (cancelled) return;
      const state = jobState(data?.status);
      if (state === "done" || state === "failed") return; // stop polling
      timer.current = window.setTimeout(tick, intervalMs);
    };

    tick();
    return () => {
      cancelled = true;
      if (timer.current) window.clearTimeout(timer.current);
    };
  }, [fetchJob, intervalMs]);

  return { job, error, history, refresh: fetchJob };
}
