import { useNavigate, useParams } from "react-router-dom";

/* ------------------------------------------------------------------ */
/*  API call — replace the body with the call your old handleSubmit   */
/*  used (from src/api or src/services). It must return the new id.   */
/* ------------------------------------------------------------------ */
type SourcePayload = {
  appName: string;
  description: string;
  repoUrl: string;
  branch: string;
};

async function submitSource(
  p: SourcePayload
): Promise<{ id: string | number }> {
  const res = await fetch(`${import.meta.env.VITE_API_BASE_URL}/jobs`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      name: p.appName,
      description: p.description || null,
      repo_url: p.repoUrl,
      branch: p.branch,
    }),
  });
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body?.detail ?? `Request failed (${res.status})`);
  }
  return res.json();
}

/* Accepts https://host/owner/repo(.git) and git@host:owner/repo(.git) */
const parseRepo = (url: string) => {
  const u = url.trim();
  const m =
    u.match(/^https?:\/\/([^/\s]+)\/(.+?)\/([^/\s]+?)(?:\.git)?\/?$/i) ??
    u.match(/^git@([^:\s]+):(.+?)\/([^/\s]+?)(?:\.git)?$/i);
  return m ? { host: m[1], owner: m[2], name: m[3] } : null;
};

const GitIcon = () => (
  <svg viewBox="0 0 24 24" aria-hidden="true">
    <path d="M10 14a3.5 3.5 0 0 0 5 0l3-3a3.5 3.5 0 0 0-5-5l-1 1" />
    <path d="M14 10a3.5 3.5 0 0 0-5 0l-3 3a3.5 3.5 0 0 0 5 5l1-1" />
  </svg>
);

const AppIcon = () => (
  <svg viewBox="0 0 24 24" aria-hidden="true">
    <rect x="3" y="4" width="18" height="16" rx="2" />
    <path d="M3 9h18M7 6.5h.01M10 6.5h.01" />
  </svg>
);

const BranchIcon = () => (
  <svg viewBox="0 0 24 24" aria-hidden="true">
    <circle cx="6" cy="5" r="2" />
    <circle cx="6" cy="19" r="2" />
    <circle cx="18" cy="7" r="2" />
    <path d="M6 7v10M18 9c0 5-12 3-12 8" />
  </svg>
);

export const SourcePage: React.FC = () => {
  const navigate = useNavigate();
  const { migrationId } = useParams();

  const [appName, setAppName] = useState("");
  const [description, setDescription] = useState("");
  const [repoUrl, setRepoUrl] = useState("");
  const [branch, setBranch] = useState("main");
  const [touched, setTouched] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const repo = useMemo(() => parseRepo(repoUrl), [repoUrl]);
  const urlInvalid = touched && repoUrl.trim() !== "" && !repo;
  const nameInvalid = touched && appName.trim() === "";
  const canSubmit =
    appName.trim() !== "" && !!repo && branch.trim() !== "" && !isLoading;

  // Fill the app name from the repo name if the user hasn't typed one
  const handleRepoBlur = () => {
    setTouched(true);
    if (!appName.trim() && repo) setAppName(repo.name);
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setTouched(true);
    if (!canSubmit) return;
    setIsLoading(true);
    setError(null);
    try {
      const created = await submitSource({
        appName: appName.trim(),
        description: description.trim(),
        repoUrl: repoUrl.trim(),
        branch: branch.trim(),
      });
      navigate(`/migration/${created.id}/scan`);
    } catch (err) {
      setError(err instanceof Error ? err.message : "Something went wrong.");
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <MainLayout
      title="Migration Skill Factory"
      subtitle="Connect a repository"
      stepNumber={1}
      totalSteps={5}
      workflow={
        <WorkflowNav
          currentStep="source"
          migrationId={migrationId}
          disabled={isLoading}
        />
      }
    >
      <section className="src-card">
        {/* Left: what happens */}
        <div className="src-intro">
          <span className="src-eyebrow">Step 1 · Source</span>
          <h2>Connect a repository</h2>
          <p>
            Point the factory at a Git repository. We pull the branch, scan the
            code and turn what we find into reusable skills for your coding
            agent.
          </p>

          <ol className="src-flow">
            <li>
              <span className="src-flow-num">1</span>
              <div>
                <strong>Pull</strong>
                <span>Clone the branch you choose</span>
              </div>
            </li>
            <li>
              <span className="src-flow-num">2</span>
              <div>
                <strong>Scan</strong>
                <span>Map packages, versions and patterns</span>
              </div>
            </li>
            <li>
              <span className="src-flow-num">3</span>
              <div>
                <strong>Generate</strong>
                <span>Skills plus a migration diff to review</span>
              </div>
            </li>
          </ol>
        </div>

        {/* Right: the form */}
        <form className="src-form" onSubmit={handleSubmit} noValidate>
          <label htmlFor="appName" className="src-label">
            Application name <span className="src-req">*</span>
          </label>
          <div className={`src-input ${nameInvalid ? "is-invalid" : ""}`}>
            <AppIcon />
            <input
              id="appName"
              type="text"
              placeholder="e.g. Customer Portal"
              value={appName}
              onChange={(e) => setAppName(e.target.value)}
              maxLength={100}
              autoComplete="off"
              disabled={isLoading}
              aria-invalid={nameInvalid}
              className="src-input-plain"
            />
          </div>
          <p className={`src-hint ${nameInvalid ? "is-error" : ""}`}>
            {nameInvalid
              ? "Application name is required."
              : "Filled from the repo name if you leave it blank."}
          </p>

          <label htmlFor="description" className="src-label">
            Description <span className="src-optional">optional</span>
          </label>
          <div className="src-input src-textarea">
            <textarea
              id="description"
              placeholder="What does this application do? Any migration notes?"
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              maxLength={500}
              rows={3}
              disabled={isLoading}
            />
          </div>
          <p className="src-hint src-counter">{description.length}/500</p>

          <div className="src-divider" />

          <label htmlFor="repoUrl" className="src-label">
            Repository URL <span className="src-req">*</span>
          </label>
          <div className={`src-input ${urlInvalid ? "is-invalid" : ""}`}>
            <GitIcon />
            <input
              id="repoUrl"
              type="text"
              placeholder="https://github.com/org/repo.git"
              value={repoUrl}
              onChange={(e) => setRepoUrl(e.target.value)}
              onBlur={handleRepoBlur}
              autoComplete="off"
              spellCheck={false}
              disabled={isLoading}
              aria-invalid={urlInvalid}
              aria-describedby="repoUrlHint"
            />
          </div>
          <p id="repoUrlHint" className={`src-hint ${urlInvalid ? "is-error" : ""}`}>
            {urlInvalid
              ? "That doesn't look like a Git URL. Use HTTPS or SSH format."
              : "HTTPS or SSH, e.g. git@github.com:org/repo.git"}
          </p>

          <label htmlFor="branch" className="src-label">
            Branch <span className="src-req">*</span>
          </label>
          <div className="src-input">
            <BranchIcon />
            <input
              id="branch"
              type="text"
              value={branch}
              onChange={(e) => setBranch(e.target.value)}
              autoComplete="off"
              spellCheck={false}
              disabled={isLoading}
            />
          </div>

          {repo && (
            <div className="src-preview">
              <span className="src-preview-host">{repo.host}</span>
              <span className="src-preview-repo">
                {repo.owner}/<strong>{repo.name}</strong>
              </span>
              <span className="src-preview-branch">@ {branch || "—"}</span>
            </div>
          )}

          {error && (
            <div className="src-error" role="alert">
              {error}
            </div>
          )}

          <button type="submit" className="src-submit" disabled={!canSubmit}>
            {isLoading ? (
              <>
                <span className="src-spinner" /> Starting…
              </>
            ) : (
              <>Start scan →</>
            )}
          </button>
        </form>
      </section>
    </MainLayout>
  );
};

export default SourcePage;
