import React, { useEffect, useMemo, useState } from "react";
import { useNavigate, useParams } from "react-router-dom";
// Keep your existing import paths for these two:
import MainLayout from "../layouts/MainLayout";
import WorkflowNav from "../components/WorkflowNav";
import { Job, jobState, useJobPolling } from "../hooks/useJobPolling";
import { FileDiff, parseDiff } from "../utils/diffParser";
import "./ReviewChangesPage.css";

const API = import.meta.env.VITE_API_BASE_URL;

type Action = "all" | "none" | "select";

const STATE_LABEL = {
  queued: "Queued",
  running: "Working",
  review: "Ready for review",
  done: "Completed",
  failed: "Failed",
} as const;

/* ---------- small pieces ---------- */

const Progress: React.FC<{ job: Job; title: string; history: { message: string }[] }> = ({ job, title, history }) => {
  const state = jobState(job.status);
  const pct = Math.max(0, Math.min(100, job.progress_percentage ?? 0));
  return (
    <section className={`rv-card rv-progress is-${state}`} aria-live="polite">
      <div className="rv-progress-top">
        <h3>{title}</h3>
        <span className={`rv-badge is-${state}`}>{STATE_LABEL[state]}</span>
      </div>
      <div className="rv-bar-row">
        <div className="rv-bar" role="progressbar" aria-valuemin={0} aria-valuemax={100} aria-valuenow={pct}>
          <div className="rv-bar-fill" style={{ width: `${pct}%` }} />
        </div>
        <span className="rv-pct">{pct}%</span>
      </div>
      <p className="rv-message">{job.progress_message || "Waiting for the worker…"}</p>
      {history.length > 1 && (
        <ol className="rv-log">
          {history.slice(0, -1).reverse().slice(0, 5).map((h, i) => <li key={i}>{h.message}</li>)}
        </ol>
      )}
    </section>
  );
};

const Checkbox: React.FC<{
  checked: boolean;
  indeterminate?: boolean;
  onChange: (v: boolean) => void;
  label: string;
  disabled?: boolean;
}> = ({ checked, indeterminate = false, onChange, label, disabled }) => (
  <input
    type="checkbox"
    className="rv-check"
    aria-label={label}
    checked={checked}
    disabled={disabled}
    ref={(el) => { if (el) el.indeterminate = indeterminate; }}
    onChange={(e) => onChange(e.target.checked)}
    onClick={(e) => e.stopPropagation()}
  />
);

const describeDecision = (d?: Job["review_decision"]) =>
  !d ? "" : d.action === "all" ? "All changes accepted"
  : d.action === "none" ? "All changes discarded"
  : `${d.accepted?.length ?? 0} hunk(s) accepted`;

/* ---------- page ---------- */

export const ReviewChangesPage: React.FC = () => {
  const navigate = useNavigate();
  const { migrationId } = useParams();

  const [pollKey, setPollKey] = useState(0);
  const { job, error, history } = useJobPolling(migrationId, 4000, pollKey);
  const state = jobState(job?.status);

  const [diffText, setDiffText] = useState<string | null>(null);
  const [summary, setSummary] = useState<string | null>(null);
  const [diffError, setDiffError] = useState<string | null>(null);
  const [selected, setSelected] = useState<Set<string>>(new Set());
  const [collapsed, setCollapsed] = useState<Set<number>>(new Set());
  const [submitting, setSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState<string | null>(null);

  // Load the diff once the coder run is waiting for review
  useEffect(() => {
    if (state !== "review" || diffText !== null || !migrationId) return;
    (async () => {
      try {
        const [d, s] = await Promise.all([
          fetch(`${API}/jobs/${migrationId}/artifacts/diff`),
          fetch(`${API}/jobs/${migrationId}/artifacts/summary`),
        ]);
        if (!d.ok) throw new Error(`Could not load the diff (${d.status})`);
        const text = await d.text();
        setDiffText(text);
        setSummary(s.ok ? await s.text() : null);
        // default: everything selected
        setSelected(new Set(parseDiff(text).flatMap((f) => f.hunks.map((h) => h.id))));
      } catch (e) {
        setDiffError(e instanceof Error ? e.message : "Could not load the diff");
      }
    })();
  }, [state, diffText, migrationId]);

  const files: FileDiff[] = useMemo(() => (diffText ? parseDiff(diffText) : []), [diffText]);
  const allIds = useMemo(() => files.flatMap((f) => f.hunks.map((h) => h.id)), [files]);
  const totals = useMemo(
    () => files.reduce((t, f) => ({ adds: t.adds + f.adds, dels: t.dels + f.dels }), { adds: 0, dels: 0 }),
    [files]
  );

  const action: Action =
    selected.size === 0 ? "none" : selected.size === allIds.length ? "all" : "select";

  const toggle = (ids: string[], on: boolean) =>
    setSelected((prev) => {
      const next = new Set(prev);
      ids.forEach((id) => (on ? next.add(id) : next.delete(id)));
      return next;
    });

  const toggleCollapse = (i: number) =>
    setCollapsed((prev) => {
      const next = new Set(prev);
      next.has(i) ? next.delete(i) : next.add(i);
      return next;
    });

  const submit = async (forced?: Action) => {
    const act = forced ?? action;
    if (act === "none" && !window.confirm("Discard all changes? Nothing will be applied to the repository.")) return;
    setSubmitting(true);
    setSubmitError(null);
    try {
      const res = await fetch(`${API}/jobs/${migrationId}/review`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ action: act, accepted: act === "select" ? [...selected] : [] }),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(typeof body?.detail === "string" ? body.detail : `Could not submit review (${res.status})`);
      }
      setPollKey((k) => k + 1); // job is queued again -> start polling
    } catch (e) {
      setSubmitError(e instanceof Error ? e.message : "Something went wrong");
    } finally {
      setSubmitting(false);
    }
  };

  const applying = !!job?.review_decision && (state === "queued" || state === "running");
  const coding = !job?.review_decision && (state === "queued" || state === "running");

  return (
    <MainLayout
      title="Migration Skill Factory"
      subtitle="Review code changes"
      stepNumber={5}
      totalSteps={5}
      workflow={<WorkflowNav currentStep="upgrade" migrationId={migrationId} disabled={submitting || applying} />}
    >
      <div className="rv-wrap">
        {error && !job && <div className="rv-alert">{error}</div>}
        {!job && !error && <div className="rv-note">Loading…</div>}

        {job && coding && <Progress job={job} history={history} title="Coder agent is working" />}
        {job && applying && <Progress job={job} history={history} title="Applying your selection" />}

        {job && state === "failed" && (
          <section className="rv-card rv-result is-failed">
            <h3>Job failed</h3>
            <p>{job.progress_message}</p>
          </section>
        )}

        {/* ---------- Done ---------- */}
        {job && state === "done" && (
          <section className="rv-card rv-result is-done">
            <span className="rv-eyebrow">Changes applied</span>
            <h3>{describeDecision(job.review_decision)}</h3>
            {job.progress_message && <p className="rv-message">{job.progress_message}</p>}
            <p className="rv-hint">
              Changes are in <span className="mono">{job.local_repo_path}</span> and are not committed yet.
            </p>
            <div className="rv-actions">
              <button className="rv-btn" onClick={() => navigate(`/migration/${job.id}/validation`)}>
                Continue to Validation →
              </button>
            </div>
          </section>
        )}

        {/* ---------- Review ---------- */}
        {job && state === "review" && (
          <>
            <section className="rv-summary">
              <div>
                <span className="rv-eyebrow">Review changes</span>
                <h2>{job.project_name}</h2>
                {job.user_query && <p className="rv-request">“{job.user_query}”</p>}
              </div>
              <div className="rv-stats">
                <div><strong>{files.length}</strong><span>files</span></div>
                <div><strong>{allIds.length}</strong><span>hunks</span></div>
                <div><strong className="rv-add">+{totals.adds}</strong><span>added</span></div>
                <div><strong className="rv-del">−{totals.dels}</strong><span>removed</span></div>
              </div>
            </section>

            {summary && (
              <details className="rv-card rv-agent-summary">
                <summary>Coder agent summary</summary>
                <pre>{summary}</pre>
              </details>
            )}

            {diffError && <div className="rv-alert">{diffError}</div>}
            {!diffText && !diffError && <div className="rv-note">Loading diff…</div>}
            {diffText !== null && files.length === 0 && (
              <div className="rv-note">The coder agent produced no changes.</div>
            )}

            {files.length > 0 && (
              <>
                <div className="rv-toolbar">
                  <span>
                    <strong>{selected.size}</strong> of {allIds.length} hunks selected
                  </span>
                  <div>
                    <button className="rv-link" onClick={() => setSelected(new Set(allIds))}>Select all</button>
                    <button className="rv-link" onClick={() => setSelected(new Set())}>Select none</button>
                    <button className="rv-link" onClick={() => setCollapsed(new Set(files.map((f) => f.index)))}>Collapse all</button>
                    <button className="rv-link" onClick={() => setCollapsed(new Set())}>Expand all</button>
                  </div>
                </div>

                {files.map((f) => {
                  const ids = f.hunks.map((h) => h.id);
                  const on = ids.filter((id) => selected.has(id)).length;
                  const isCollapsed = collapsed.has(f.index);
                  return (
                    <section key={f.index} className="rv-file">
                      <header className="rv-file-head" onClick={() => toggleCollapse(f.index)}>
                        <Checkbox
                          label={`Select all hunks in ${f.path}`}
                          checked={ids.length > 0 && on === ids.length}
                          indeterminate={on > 0 && on < ids.length}
                          disabled={ids.length === 0 || submitting}
                          onChange={(v) => toggle(ids, v)}
                        />
                        <span className="rv-caret">{isCollapsed ? "▸" : "▾"}</span>
                        <span className="mono rv-path">{f.path}</span>
                        {f.isNew && <span className="rv-tag is-new">new</span>}
                        {f.isDeleted && <span className="rv-tag is-del">deleted</span>}
                        {f.isBinary && <span className="rv-tag">binary</span>}
                        <span className="rv-counts">
                          <span className="rv-add">+{f.adds}</span> <span className="rv-del">−{f.dels}</span>
                        </span>
                      </header>

                      {!isCollapsed &&
                        f.hunks.map((h) => {
                          const isOn = selected.has(h.id);
                          return (
                            <div key={h.id} className={`rv-hunk ${isOn ? "" : "is-off"}`}>
                              <div className="rv-hunk-head">
                                <Checkbox
                                  label={`Accept hunk ${h.id}`}
                                  checked={isOn}
                                  disabled={submitting}
                                  onChange={(v) => toggle([h.id], v)}
                                />
                                <span className="mono">{h.header}</span>
                                <span className="rv-hunk-id mono">{h.id}</span>
                              </div>
                              <table className="rv-lines">
                                <tbody>
                                  {h.lines.map((l, i) => (
                                    <tr key={i} className={`is-${l.type}`}>
                                      <td className="rv-no">{l.oldNo ?? ""}</td>
                                      <td className="rv-no">{l.newNo ?? ""}</td>
                                      <td className="rv-sign">{l.type === "add" ? "+" : l.type === "del" ? "−" : ""}</td>
                                      <td className="rv-code">{l.text || " "}</td>
                                    </tr>
                                  ))}
                                </tbody>
                              </table>
                            </div>
                          );
                        })}
                    </section>
                  );
                })}
              </>
            )}

            {/* ---------- Decision bar ---------- */}
            <section className="rv-footer">
              <div className="rv-hint">
                {action === "all" && "All changes will be applied."}
                {action === "none" && "Nothing selected: all changes will be discarded."}
                {action === "select" && `${selected.size} of ${allIds.length} hunks will be applied, the rest discarded.`}
              </div>
              {submitError && <div className="rv-alert">{submitError}</div>}
              <div className="rv-actions">
                <button className="rv-btn is-danger" disabled={submitting} onClick={() => submit("none")}>
                  Discard all
                </button>
                <button className="rv-btn" disabled={submitting || diffText === null} onClick={() => submit()}>
                  {submitting
                    ? "Submitting…"
                    : action === "all"
                    ? "Accept all changes"
                    : action === "none"
                    ? "Discard all changes"
                    : `Apply ${selected.size} selected`}
                </button>
              </div>
            </section>
          </>
        )}
      </div>
    </MainLayout>
  );
};

export default ReviewChangesPage;
